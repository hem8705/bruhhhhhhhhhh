import pandas as pd
import numpy as np
from sklearn.tree import DecisionTreeClassifier
from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.svm import SVC
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score
from sklearn.ensemble import RandomForestClassifier, RandomForestRegressor
from sklearn.neighbors import KNeighborsClassifier
from sklearn.neural_network import MLPClassifier
#import matplotlib.pyplot as plt
df = pd.read_csv('cleveland.csv', header = None)

df.columns = ['age', 'sex', 'cp', 'trestbps', 'chol',
              'fbs', 'restecg', 'thalach', 'exang',
              'oldpeak', 'slope', 'ca', 'thal', 'target']
df.isnull().sum()
df.dropna(inplace=True)

df['target'] = df.target.map({0: 0, 1: 1, 2: 1, 3: 1, 4: 1})
df['sex'] = df.sex.map({0: 'female', 1: 'male'})

df['ca'] = df.ca.fillna(df.ca.mean())

df.fillna(0, inplace=True)
X = df.iloc[:, :-1].values
Y= df.iloc[:, -1].values
thedata=pd.read_csv("Heart_Disease_Prediction.csv")

       #'EKG results', 'Max HR', 'Exercise angina', 'ST depression',
       #'Slope of ST', 'Number of vessels fluro', 'Thallium']])

Y=Y.ravel()


svc_predictor=make_pipeline(StandardScaler(),SVC(gamma="auto"))
rf_predictor=RandomForestClassifier(n_estimators=60)

X_train, X_test, y_train, y_test = train_test_split(X, Y, test_size=0.33, random_state=40)
svc_predictor.fit(X_train, y_train)
rf_predictor.fit(X_train,y_train)
y_pred_rf=rf_predictor.predict(X_test)
y_predicted_svc = svc_predictor.predict(X_test)
print("SVC:",accuracy_score(y_test,y_predicted_svc))
print("Random Forest:",accuracy_score(y_test,y_pred_rf))
aggregate_results=pd.DataFrame(y_pred_rf)
aggregate_results.columns=["First"]
decision_tree=DecisionTreeClassifier()
neigh=KNeighborsClassifier(n_neighbors=5)
neigh.fit(X_train,y_train)
y_pred_knn=neigh.predict(X_test)
print("KNN:",accuracy_score(y_test,y_pred_knn))
decision_tree.fit(X_train,y_train)
y_pred_dt=decision_tree.predict(X_test)
print("Decision Tree:",accuracy_score(y_test,y_pred_dt))

mlp = MLPClassifier(hidden_layer_sizes=(6, 2), random_state=1)
mlp.fit(X_train,y_train)
y_pred_mlp=mlp.predict(X_test)
print("MLP:",accuracy_score(y_test,y_pred_mlp))
for counter in range(0,50):
    rf_predictor = RandomForestClassifier(n_estimators=60)
    rf_predictor.fit(X_train, y_train)
    y_pred_rf = rf_predictor.predict(X_test)
    y_predicted_df=pd.DataFrame(y_pred_rf)

    aggregate_results=aggregate_results.join(y_predicted_df[counter])
